\querytitle{1}
\querymeta{DDL}{Create the pricing slab table with primary-key, unique, foreign-key, and CHECK constraints so invalid or overlapping weight boundaries are rejected at the database layer.}

\textbf{SQL:}
\begin{lstlisting}
CREATE TABLE IF NOT EXISTS pricing_slabs (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    pricing_lane_id BIGINT UNSIGNED NOT NULL,
    min_weight_grams INT UNSIGNED NOT NULL,
    max_weight_grams INT UNSIGNED NOT NULL,
    base_fee DECIMAL(12,2) NOT NULL DEFAULT 0,
    extra_kg_fee DECIMAL(12,2) NOT NULL DEFAULT 0,
    cod_percentage DECIMAL(5,2) NOT NULL DEFAULT 0,
    cod_fixed_fee DECIMAL(12,2) NOT NULL DEFAULT 0,
    status ENUM('ACTIVE','INACTIVE') NOT NULL DEFAULT 'ACTIVE',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_pricing_slabs_lane_min
        (pricing_lane_id, min_weight_grams),
    UNIQUE KEY uq_pricing_slabs_lane_max
        (pricing_lane_id, max_weight_grams),
    CONSTRAINT fk_pricing_slabs_lane
        FOREIGN KEY (pricing_lane_id) REFERENCES pricing_lanes(id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT chk_pricing_slabs_range
        CHECK (max_weight_grams > min_weight_grams)
);
\end{lstlisting}

\querytitle{10}
\querymeta{JOIN}{Combine parcel, sender, origin-hub, and destination-hub data into one operations report.}

\textbf{SQL:}
\begin{lstlisting}
SELECT
    p.tracking_number,
    c.name AS sender_name,
    oh.name AS origin_hub,
    dh.name AS destination_hub,
    p.status
FROM parcels AS p
INNER JOIN customers AS c
    ON c.id = p.sender_customer_id
INNER JOIN hubs AS oh
    ON oh.id = p.origin_hub_id
INNER JOIN hubs AS dh
    ON dh.id = p.destination_hub_id
ORDER BY p.created_at DESC;
\end{lstlisting}

\querytitle{11}
\querymeta{Aggregate-GROUP BY}{Summarize parcel count, total delivery fees, and total COD value for each parcel status.}

\textbf{SQL:}
\begin{lstlisting}
SELECT
    status,
    COUNT(*) AS parcel_count,
    SUM(delivery_fee) AS total_delivery_fees,
    SUM(cod_amount) AS total_cod_value
FROM parcels
GROUP BY status
ORDER BY parcel_count DESC, status ASC;
\end{lstlisting}
\queryoutput{11}

\querytitle{12}
\querymeta{Subquery}{Identify parcels whose delivery fee is higher than the average fee across all parcels.}

\textbf{SQL:}
\begin{lstlisting}
SELECT tracking_number, delivery_fee, status
FROM parcels
WHERE delivery_fee > (
    SELECT AVG(delivery_fee)
    FROM parcels
)
ORDER BY delivery_fee DESC;
\end{lstlisting}
